cmake_minimum_required(VERSION 3.28)

# vcpkg evaluates the manifest in project(): choose features beforehand.
option(TCHIPS_BUILD_CLIENT "Build the native SDL3 client" ON)
option(TCHIPS_BUILD_SERVER "Build the headless server" ON)
option(TCHIPS_BUILD_TESTS "Build GoogleTest and CTest checks" ON)
option(TCHIPS_ENABLE_NETWORK "Build GameNetworkingSockets and Protobuf networking" ON)
option(TCHIPS_ENABLE_UI "Resolve RmlUi (no renderer adapter yet)" OFF)
option(TCHIPS_ENABLE_AUDIO "Resolve miniaudio (no audio engine yet)" OFF)
option(TCHIPS_ENABLE_PERSISTENCE "Resolve libpqxx (no database connection yet)" OFF)
option(TCHIPS_ENABLE_PROFILING "Instrument executables with Tracy" OFF)
set(VCPKG_MANIFEST_NO_DEFAULT_FEATURES ON CACHE BOOL "Explicit phase features" FORCE)
set(VCPKG_MANIFEST_FEATURES "")
foreach(pair IN ITEMS "CLIENT;client" "TESTS;tests")
  list(GET pair 0 flag)
  list(GET pair 1 feature)
  if(TCHIPS_BUILD_${flag})
    list(APPEND VCPKG_MANIFEST_FEATURES ${feature})
  endif()
endforeach()
foreach(feature IN ITEMS network ui audio persistence profiling)
  string(TOUPPER "${feature}" flag)
  if(TCHIPS_ENABLE_${flag})
    list(APPEND VCPKG_MANIFEST_FEATURES ${feature})
  endif()
endforeach()

project(TchipsNative VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(cmake/ProjectOptions.cmake)
include(cmake/Dependencies.cmake)
add_subdirectory(shared)
if(TCHIPS_BUILD_CLIENT OR TCHIPS_BUILD_SERVER)
  add_custom_target(tchips_map_assets ALL
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/assets/maps"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PROJECT_SOURCE_DIR}/content/maps/prototype.map" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/assets/maps/prototype.map"
    VERBATIM)
endif()
if(TCHIPS_ENABLE_NETWORK)
  add_subdirectory(protocol)
endif()
if(TCHIPS_BUILD_CLIENT OR TCHIPS_BUILD_TESTS)
  add_subdirectory(apps/client)
endif()
if(TCHIPS_BUILD_SERVER)
  add_subdirectory(apps/server)
endif()
if(TCHIPS_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
